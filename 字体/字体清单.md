## 简短复核代码

以下 Python 代码需在已有 fontTools 的环境中运行，只读取字体与清单；哈希比对覆盖字体和许可证，字符检查会打印实际缺字。

```python
from pathlib import Path
import hashlib, re
from fontTools.ttLib import TTFont
root = Path('/Users/limingyang/Desktop/单依纯二专/字体')
chars = '纯妹妹×Clawd还有什么更好的向日葵朝着夜假如我们还爱着我管不住我自己有趣表示理解你茉莉多少光阴珠玉0123456789'
report = (root / "字体清单.md").read_text()
for path in sorted(root.glob("*/*")):
    assert path.suffix in {".ttf", ".otf"} or path.name == "OFL.txt"
    section = report.split(f"](<{path}>)", 1)[1]
    expected = re.search(r"SHA256：`([0-9a-f]{64})`", section).group(1)
    assert hashlib.sha256(path.read_bytes()).hexdigest() == expected
    if path.suffix in {".ttf", ".otf"}:
        with TTFont(path, checkChecksums=2) as font:
            cmap = font.getBestCmap()
            missing = "".join(c for c in dict.fromkeys(chars) if ord(c) not in cmap or font.getGlyphID(cmap[ord(c)]) == 0)
            print(path.name, missing or "全部包含")
```
